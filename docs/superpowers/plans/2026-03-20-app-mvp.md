import { create } from "zustand";
import AsyncStorage from "@react-native-async-storage/async-storage";
import * as SecureStore from "expo-secure-store";
import { FREE_DAILY_CALLS } from "../constants/theme";

// --- Auth Store ---
interface AuthState {
  jwt: string | null;
  walletAddress: string | null;
  isConnected: boolean;
  setAuth: (jwt: string, wallet: string) => void;
  clearAuth: () => void;
  loadAuth: () => Promise<void>;
}

export const useAuthStore = create<AuthState>((set) => ({
  jwt: null,
  walletAddress: null,
  isConnected: false,
  setAuth: async (jwt, wallet) => {
    await SecureStore.setItemAsync("jwt", jwt);
    await SecureStore.setItemAsync("wallet", wallet);
    set({ jwt, walletAddress: wallet, isConnected: true });
  },
  clearAuth: async () => {
    await SecureStore.deleteItemAsync("jwt");
    await SecureStore.deleteItemAsync("wallet");
    set({ jwt: null, walletAddress: null, isConnected: false });
  },
  loadAuth: async () => {
    const jwt = await SecureStore.getItemAsync("jwt");
    const wallet = await SecureStore.getItemAsync("wallet");
    if (jwt && wallet) set({ jwt, walletAddress: wallet, isConnected: true });
  },
}));

// --- Chat Store ---
export interface Message {
  id: string;
  role: "user" | "assistant";
  content: string;
  timestamp: number;
}

export interface Conversation {
  id: string;
  title: string;
  messages: Message[];
  model: string;
  createdAt: number;
  updatedAt: number;
}

interface ChatState {
  conversations: Conversation[];
  loadConversations: () => Promise<void>;
  saveConversation: (conv: Conversation) => Promise<void>;
  deleteConversation: (id: string) => Promise<void>;
}

export const useChatStore = create<ChatState>((set, get) => ({
  conversations: [],
  loadConversations: async () => {
    const raw = await AsyncStorage.getItem("conversations");
    if (raw) set({ conversations: JSON.parse(raw) });
  },
  saveConversation: async (conv) => {
    const existing = get().conversations.filter((c) => c.id !== conv.id);
    const updated = [conv, ...existing].slice(0, 100);
    set({ conversations: updated });
    await AsyncStorage.setItem("conversations", JSON.stringify(updated));
  },
  deleteConversation: async (id) => {
    const updated = get().conversations.filter((c) => c.id !== id);
    set({ conversations: updated });
    await AsyncStorage.setItem("conversations", JSON.stringify(updated));
  },
}));

// --- Free Tier Store ---
interface FreeTierState {
  callsUsedToday: number;
  lastResetDate: string;
  callsRemaining: number;
  useCall: () => boolean; // returns true if call allowed
  addBonusCalls: (n: number) => void;
  loadFreeTier: () => Promise<void>;
}

const todayStr = () => new Date().toISOString().slice(0, 10);

export const useFreeTierStore = create<FreeTierState>((set, get) => ({
  callsUsedToday: 0,
  lastResetDate: todayStr(),
  callsRemaining: FREE_DAILY_CALLS,
  useCall: () => {
    const state = get();
    let used = state.callsUsedToday;
    let resetDate = state.lastResetDate;
    if (resetDate !== todayStr()) {
      used = 0;
      resetDate = todayStr();
    }
    if (used >= state.callsRemaining) return false;
    used++;
    set({ callsUsedToday: used, lastResetDate: resetDate });
    AsyncStorage.setItem("freeTier", JSON.stringify({ used, resetDate, bonus: state.callsRemaining - FREE_DAILY_CALLS }));
    return true;
  },
  addBonusCalls: (n) => {
    set((s) => ({ callsRemaining: s.callsRemaining + n }));
  },
  loadFreeTier: async () => {
    const raw = await AsyncStorage.getItem("freeTier");
    if (raw) {
      const { used, resetDate, bonus } = JSON.parse(raw);
      if (resetDate === todayStr()) {
        set({ callsUsedToday: used, lastResetDate: resetDate, callsRemaining: FREE_DAILY_CALLS + (bonus || 0) });
      }
    }
  },
}));

// --- Preferences Store ---
interface PrefsState {
  theme: "dark" | "light";
  defaultModel: string;
  developerMode: boolean;
  setTheme: (t: "dark" | "light") => void;
  setDefaultModel: (m: string) => void;
  setDeveloperMode: (d: boolean) => void;
  loadPrefs: () => Promise<void>;
}

export const usePrefsStore = create<PrefsState>((set) => ({
  theme: "dark",
  defaultModel: "claude-haiku",
  developerMode: false,
  setTheme: (theme) => { set({ theme }); AsyncStorage.setItem("prefs_theme", theme); },
  setDefaultModel: (m) => { set({ defaultModel: m }); AsyncStorage.setItem("prefs_model", m); },
  setDeveloperMode: (d) => { set({ developerMode: d }); AsyncStorage.setItem("prefs_dev", String(d)); },
  loadPrefs: async () => {
    const theme = (await AsyncStorage.getItem("prefs_theme")) as "dark" | "light" | null;
    const model = await AsyncStorage.getItem("prefs_model");
    const dev = await AsyncStorage.getItem("prefs_dev");
    set({
      theme: theme || "dark",
      defaultModel: model || "claude-haiku",
      developerMode: dev === "true",
    });
  },
}));

Create `lib/api.ts`:
```ts
import { API_BASE } from "../constants/theme";
import { useAuthStore } from "./store";

class ApiClient {
  private getHeaders(): Record<string, string> {
    const headers: Record<string, string> = { "Content-Type": "application/json" };
    const jwt = useAuthStore.getState().jwt;
    if (jwt) headers["Authorization"] = `Bearer ${jwt}`;
    return headers;
  }

  async get<T>(path: string): Promise<T> {
    const res = await fetch(`${API_BASE}${path}`, { headers: this.getHeaders() });
    if (!res.ok) throw new ApiError(res.status, await res.text());
    return res.json();
  }

  async post<T>(path: string, body: Record<string, unknown>): Promise<T> {
    const res = await fetch(`${API_BASE}${path}`, {
      method: "POST",
      headers: this.getHeaders(),
      body: JSON.stringify(body),
    });
    if (res.status === 402) {
      throw new PaymentRequiredError(res.headers);
    }
    if (!res.ok) throw new ApiError(res.status, await res.text());
    return res.json();
  }

  streamChat(
    messages: { role: string; content: string }[],
    model: string,
    onToken: (text: string) => void,
    onDone: (meta: { model: string; cost_usd: number }) => void,
    onError: (err: Error) => void,
  ): AbortController {
    const controller = new AbortController();
    const headers = this.getHeaders();
    headers["Accept"] = "text/event-stream";

    fetch(`${API_BASE}/stream/chat`, {
      method: "POST",
      headers,
      body: JSON.stringify({ messages, model, max_tokens: 2048 }),
      signal: controller.signal,
    })
      .then(async (res) => {
        if (res.status === 402) {
          onError(new PaymentRequiredError(res.headers));
          return;
        }
        if (!res.ok) {
          onError(new ApiError(res.status, await res.text()));
          return;
        }
        const reader = res.body?.getReader();
        if (!reader) return;
        const decoder = new TextDecoder();
        let buffer = "";
        while (true) {
          const { done, value } = await reader.read();
          if (done) break;
          buffer += decoder.decode(value, { stream: true });
          const lines = buffer.split("\n");
          buffer = lines.pop() || "";
          for (const line of lines) {
            if (!line.startsWith("data: ")) continue;
            const data = JSON.parse(line.slice(6));
            if (data.done) {
              onDone({ model: data.model, cost_usd: data.cost_usd });
            } else if (data.text) {
              onToken(data.text);
            }
          }
        }
      })
      .catch((err) => {
        if (err.name !== "AbortError") onError(err);
      });

    return controller;
  }
}

export class ApiError extends Error {
  constructor(public status: number, public body: string) {
    super(`API ${status}: ${body.slice(0, 200)}`);
  }
}

export class PaymentRequiredError extends Error {
  public paymentHeaders: Headers;
  constructor(headers: Headers) {
    super("Payment required (402)");
    this.paymentHeaders = headers;
  }
}

export const api = new ApiClient();
```

import { useEffect, useState } from "react";

interface Props {
  text: string;
  isStreaming: boolean;

export function StreamingText({ text, isStreaming }: Props) {
  const [showCursor, setShowCursor] = useState(true);

  useEffect(() => {
    if (!isStreaming) { setShowCursor(false); return; }
    const interval = setInterval(() => setShowCursor((v) => !v), 500);
    return () => clearInterval(interval);
  }, [isStreaming]);

  return (
    <View className="max-w-[85%] self-start mb-3">
      <View className="bg-bg-card px-4 py-3 rounded-2xl rounded-bl-sm">

        setStreaming("");

import { View, Text, TextInput, Pressable, ScrollView, ActivityIndicator } from "react-native";

import { ArrowLeft, Play } from "lucide-react-native";
import { useRouter } from "expo-router";
